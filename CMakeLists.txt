cmake_minimum_required(VERSION 3.24)
project(lmfinance VERSION 0.1.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Third-party libraries are static; only system glibc may be dynamic on Linux.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()

foreach(dep quantlib boost)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dep}")
    message(FATAL_ERROR "third_party/${dep} missing: run scripts/sync-deps.sh")
  endif()
endforeach()

# Embed schema and notices.
function(embed_file var file ns name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
embed_file(SCHEMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmfinance.dsl.json" lmfinance embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmfinance { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" lmfinance embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmfinance { ${NOTICES_SRC} }\n")

# QuantLib through its own CMake (it generates ql/config.hpp), without tests, examples, install or
# the experimental/legacy/market-model trees that lmfinance does not use. Boost is header-only here.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt QL_BUILD_EXAMPLES QL_BUILD_TEST_SUITE QL_BUILD_BENCHMARK QL_BUILD_FUZZ_TEST_SUITE QL_INSTALL_BENCHMARK
        QL_INSTALL_EXAMPLES QL_INSTALL_TEST_SUITE QL_INSTALL_ENABLE QL_ENABLE_OPENMP)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(Boost_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost" CACHE PATH "" FORCE)
set(Boost_NO_SYSTEM_PATHS ON CACHE BOOL "" FORCE)
set(Boost_NO_BOOST_CMAKE ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF)
# Compile time matters more than the last 10% of speed: -O1 for QuantLib, -O2 for the host code.
set(CMAKE_CXX_FLAGS_RELEASE "-O1 -DNDEBUG")
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)  # QuantLib uses FindBoost with header-only Boost
endif()
add_subdirectory(third_party/quantlib EXCLUDE_FROM_ALL)
get_target_property(QL_SRCS ql_library SOURCES)
# experimental/coupons stays: the core coupon pricers refer to CmsSpreadCoupon.
set(KEEP_SRCS "")
foreach(src IN LISTS QL_SRCS)
  if(src MATCHES "(^|/)experimental/" AND NOT src MATCHES "experimental/coupons/")
    continue()
  endif()
  if(src MATCHES "(^|/)legacy/|marketmodels/")
    continue()
  endif()
  list(APPEND KEEP_SRCS "${src}")
endforeach()
set(QL_SRCS ${KEEP_SRCS})
set_property(TARGET ql_library PROPERTY SOURCES ${QL_SRCS})
target_compile_options(ql_library PRIVATE -w)
# QuantLib exports its include dirs for install(); CMake rejects a Boost path inside our source tree.
foreach(prop INTERFACE_INCLUDE_DIRECTORIES INTERFACE_SYSTEM_INCLUDE_DIRECTORIES)
  get_target_property(dirs ql_library ${prop})
  list(FILTER dirs EXCLUDE REGEX "third_party/boost")
  set_property(TARGET ql_library PROPERTY ${prop} ${dirs})
endforeach()
target_include_directories(ql_library SYSTEM PRIVATE third_party/boost)

add_executable(lmfinance cpp/main.cpp cpp/finance.cpp cpp/memory_budget.cpp)
target_compile_definitions(lmfinance PRIVATE LMFINANCE_VERSION="${PROJECT_VERSION}")
target_include_directories(lmfinance PRIVATE cpp "${GEN}")
target_include_directories(lmfinance SYSTEM PRIVATE third_party/boost)
target_link_libraries(lmfinance PRIVATE ql_library)
if(NOT MSVC)
  target_compile_options(lmfinance PRIVATE -O2 -Wall -Wextra)
endif()
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmfinance PRIVATE -static-libstdc++ -static-libgcc)
  if(WIN32)
    target_link_options(lmfinance PRIVATE -static)
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(lmfinance PRIVATE -s)
  endif()
endif()
